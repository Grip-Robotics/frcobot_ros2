cmake_minimum_required(VERSION 3.16)
project(fairino_servo_offline_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
if(NOT EXISTS "${GOOGLETEST_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set GOOGLETEST_SOURCE_DIR to a local GoogleTest checkout; no downloads are made")
endif()
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${GOOGLETEST_SOURCE_DIR}" googletest)
get_filename_component(PACKAGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(servo_j_streamer "${PACKAGE_ROOT}/src/servo_j_streamer.cpp")
target_include_directories(servo_j_streamer PUBLIC "${PACKAGE_ROOT}/include")
target_link_libraries(servo_j_streamer PUBLIC Threads::Threads)
foreach(name servo_j_streamer servo_j_udp_diagnostics servo_start_settings servo_start_sdk controller_state_progress)
  add_executable(test_${name} "${PACKAGE_ROOT}/test/test_${name}.cpp")
  target_link_libraries(test_${name} PRIVATE servo_j_streamer GTest::gtest_main)
  add_test(NAME test_${name} COMMAND test_${name})
endforeach()
target_include_directories(test_servo_start_sdk PRIVATE "${PACKAGE_ROOT}")
target_link_libraries(test_servo_start_sdk PRIVATE
  "${PACKAGE_ROOT}/libfairino/lib/libfairino.so.2.4.1" "-Wl,--export-dynamic")
target_sources(test_servo_start_sdk PRIVATE "${PACKAGE_ROOT}/test/offline_network_guard.cpp")
add_executable(test_servo_j_sdk_241 "${PACKAGE_ROOT}/test/test_servo_j_sdk.cpp"
  "${PACKAGE_ROOT}/test/offline_network_guard.cpp")
target_include_directories(test_servo_j_sdk_241 PRIVATE "${PACKAGE_ROOT}")
target_link_libraries(test_servo_j_sdk_241 PRIVATE GTest::gtest_main
  "${PACKAGE_ROOT}/libfairino/lib/libfairino.so.2.4.1" "-Wl,--export-dynamic")
add_test(NAME test_servo_j_sdk_241 COMMAND test_servo_j_sdk_241)
if(SDK234_PACKAGE_ROOT)
  add_executable(test_servo_j_sdk_234 "${PACKAGE_ROOT}/test/test_servo_j_sdk.cpp"
    "${PACKAGE_ROOT}/test/offline_network_guard.cpp")
  target_compile_definitions(test_servo_j_sdk_234 PRIVATE FAIRINO_SDK_234)
  target_include_directories(test_servo_j_sdk_234 PRIVATE "${SDK234_PACKAGE_ROOT}")
  target_link_libraries(test_servo_j_sdk_234 PRIVATE GTest::gtest_main
    "${SDK234_PACKAGE_ROOT}/libfairino/lib/libfairino.so.2.3.4" "-Wl,--export-dynamic")
  add_test(NAME test_servo_j_sdk_234 COMMAND test_servo_j_sdk_234)
endif()
